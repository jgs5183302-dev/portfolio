 /* ==================================================

    const snapContainer = document.querySelector('.snap-container');
    const sections = document.querySelectorAll('.section');
    const personImg = document.getElementById('fixedPersonImg');
    const starsImg = document.getElementById('fixedStarsImg');
    const navLinks = document.querySelectorAll('.nav-bar a');
  
    // 하단 슬라이더 전용 모달 (FLIP 모션용)
    const flyCard = document.getElementById('flyCard');
    const flyTitle = document.getElementById('flyProjectTitle');
    const closeFlyBtn = document.getElementById('closeFlyBtn');
    const btnFlyPdf = document.getElementById('btnFlyPdf');
    const btnFlyVideo = document.getElementById('btnFlyVideo');
  
    // 전체화면 뷰어 (공용)
    const fullViewer = document.getElementById('fullViewer');
    const viewerBox = document.getElementById('viewerBox');
    const closeViewerBtn = document.getElementById('closeViewerBtn');
  
    let activePdf = '';
    let activeVideo = '';
    let lastClickedRect = null;
  
    /* ==================================================
       2. 네비게이션 단일 활성화 & 쫀득한 휠 스냅
       ================================================== */
    let currentIndex = 0;
    let isThrottled = false;
  
    function updateNav(targetIndex) {
      navLinks.forEach((link, idx) => {
        if (idx === targetIndex) {
          link.classList.add('active');
        } else {
          link.classList.remove('active');
        }
      });
    }
  
    updateNav(0);
  
  /* ==================================================
       네비게이션 & 스크롤 & 미디어 통합 제어
       ================================================== */
       function moveToSection(index) {
      if (index < 0 || index >= sections.length) return;
      currentIndex = index;

      sections[currentIndex].scrollIntoView({ behavior: 'smooth' });
      updateNav(currentIndex);

      // 2페이지 쇼케이스 영상 제어
      if (showcaseVideo) {
        if (currentIndex === 1) {
          showcaseVideo.play().catch(() => {});
        } else {
          showcaseVideo.pause();
        }
      }

      // 페이지 이동 시 BGM 자동 동기화 (2페이지에선 정지, 타 페이지에선 재생)
      if (typeof syncBgmPlayback === 'function') {
        syncBgmPlayback();
      }

      // 1페이지 배경 비디오 및 별 배경 제어
      if (currentIndex > 0) {
        if (personImg) {
          personImg.classList.add('blurred');
          if (typeof personImg.pause === 'function') personImg.pause();
        }
        if (starsImg) starsImg.classList.add('active');
      } else {
        if (personImg) {
          personImg.classList.remove('blurred');
          if (typeof personImg.play === 'function') personImg.play().catch(() => {});
        }
        if (starsImg) starsImg.classList.remove('active');
      }

      setTimeout(() => {
        isThrottled = false;
      }, 550);
    }

    // 마우스 휠 스크롤 제어
    window.addEventListener('wheel', (e) => {
      if (fullViewer && fullViewer.style.display === 'flex') {
        return;
      }

      if (isThrottled) return;

      if (e.deltaY > 10) {
        if (currentIndex < sections.length - 1) {
          isThrottled = true;
          moveToSection(currentIndex + 1);
        }
      } else if (e.deltaY < -10) {
        if (currentIndex > 0) {
          isThrottled = true;
          moveToSection(currentIndex - 1);
        }
      }
    }, { passive: true });

    // 상단 네비게이션 클릭 시 해당 섹션으로 이동
    navLinks.forEach((link, idx) => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        moveToSection(idx);
      });
    });

    // 스냅 컨테이너 스크롤 감지 및 실시간 동기화
    if (snapContainer) {
      snapContainer.addEventListener('scroll', () => {
        const centerY = snapContainer.scrollTop + window.innerHeight / 2;
        let activeIdx = 0;

        sections.forEach((section, idx) => {
          const top = section.offsetTop;
          const bottom = top + section.offsetHeight;
          if (centerY >= top && centerY < bottom) {
            activeIdx = idx;
          }
        });

        if (currentIndex !== activeIdx) {
          currentIndex = activeIdx;
          updateNav(activeIdx);

          if (showcaseVideo) {
            if (currentIndex === 1) {
              showcaseVideo.play().catch(() => {});
            } else {
              showcaseVideo.pause();
            }
          }

          // 스크롤로 섹션 넘어갈 때 BGM 상태 동기화
          if (typeof syncBgmPlayback === 'function') {
            syncBgmPlayback();
          }
        }

        if (currentIndex > 0) {
          if (personImg) {
            personImg.classList.add('blurred');
            if (typeof personImg.pause === 'function') personImg.pause();
          }
          if (starsImg) starsImg.classList.add('active');
        } else {
          if (personImg) {
            personImg.classList.remove('blurred');
            if (typeof personImg.play === 'function') personImg.play().catch(() => {});
          }
          if (starsImg) starsImg.classList.remove('active');
        }
      });
    }
  
    /* ==================================================
       3. 2페이지 쇼케이스: 영상 전환 / 리스트 클릭 / 전체화면

    const portfolioData = [
      {
        title: "1. 에뛰드 하우스 리브랜딩",

        video: "에뛰드하우스_수정본.mp4",

      },
      {
        title: "2. 아이엠스타 예고편",
        caption: "2. 아이엠스타 예고편: 아이엠스타 극장판(가상)의 예고편으로 추억의 애니메이션의 화려한 복귀를 보여준다.",
        video: "아이엠스타.mp4",
        storyboard: "아이엠스타_스토리보드(수정).pdf"
      },
      {
        title: "3. Project 03",
        video: "",
        storyboard: ""
      },
      {
        title: "4. Project 04",
        video: "",
        storyboard: ""
      }
    ];
/* ==================================================
     사운드 토글 & 배경음악(BGM) 정밀 제어
     ================================================== */
     const bgMusic = document.getElementById('bgMusicPlayer');
  const bgmBtn = document.getElementById('bgmToggleBtn');
  const iconMuted = bgmBtn ? bgmBtn.querySelector('.icon-muted') : null;
  const iconPlaying = bgmBtn ? bgmBtn.querySelector('.icon-playing') : null;

  let isBgmActive = false;

  function updateBgmUI(on) {
    if (!bgmBtn) return;
    if (on) {
      bgmBtn.classList.add('active');
      if (iconMuted) iconMuted.style.display = 'none';
      if (iconPlaying) iconPlaying.style.display = 'inline';
    } else {
      bgmBtn.classList.remove('active');
      if (iconMuted) iconMuted.style.display = 'inline';
      if (iconPlaying) iconPlaying.style.display = 'none';
    }
  }

  // BGM 재생 시도 함수
  function playAudio() {
    if (!bgMusic) return;

    bgMusic.volume = 0.5; // 50% 볼륨
    const promise = bgMusic.play();

    if (promise !== undefined) {
      promise.then(() => {
        isBgmActive = true;
        updateBgmUI(true);
      }).catch((err) => {
        isBgmActive = false;
        updateBgmUI(false);
        alert("소리 재생 실패 원인: " + err.message + "\n(파일 경로가 정확한지, 혹은 mp3 파일인지 확인해주세요)");
      });
    }
  }

  // BGM 일시정지
  function pauseAudio() {
    if (!bgMusic) return;
    bgMusic.pause();
    isBgmActive = false;
    updateBgmUI(false);
  }

  // 상단 BGM 버튼 클릭 시 실행
  if (bgmBtn && bgMusic) {
    bgmBtn.addEventListener('click', (e) => {
      e.stopPropagation();

      if (bgMusic.paused) {
        playAudio();
      } else {
        pauseAudio();
      }
    });
  }

  // 쇼케이스 메인 비디오(2페이지)와 상호작용
  if (showcaseVideo && bgMusic) {
    showcaseVideo.addEventListener('play', () => {
      // 메인 영상이 재생되면 배경음악 일시정지
      if (!bgMusic.paused) {
        bgMusic.pause();
        updateBgmUI(false);
      }
    });

    showcaseVideo.addEventListener('pause', () => {
      // 2페이지를 벗어났거나 팝업이 닫힌 경우 다시 BGM 재생
      if (isBgmActive && (!fullViewer || fullViewer.style.display !== 'flex')) {
        playAudio();
      }
    });
  }
    let currentProjectIdx = 0;
    const showcaseVideo = document.getElementById('portfolioMainVideo');
    const showcaseCaption = document.getElementById('workCaptionText');
    const prevBtn = document.getElementById('prevWorkBtn');
    const nextBtn = document.getElementById('nextWorkBtn');
    const projectItems = document.querySelectorAll('.project-item');
    const mainViewTrigger = document.getElementById('mainViewTrigger');
  
    // 영상 및 리스트 상태 변경 함수
    function updateArrowVisibility() {
      if (!prevBtn) return;
      
      // 첫 번째 프로젝트(인덱스 0)일 때만 왼쪽 화살표 숨김
      if (currentProjectIdx === 0) {
        prevBtn.classList.add('disabled');
      } else {
        prevBtn.classList.remove('disabled');
      }
    }
  
    function playProject(index) {
      if (index < 0 || index >= portfolioData.length) return;
      if (!portfolioData[index].video) {
        alert('등록된 영상 파일이 없습니다.');
        return;
      }
  
      currentProjectIdx = index;
      const project = portfolioData[currentProjectIdx];
  
      // 1. 비디오 소스 교체 및 재생
      if (showcaseVideo) {
        showcaseVideo.src = encodeURI(project.video);
        showcaseVideo.load();
        showcaseVideo.play().catch(() => {});
      }
  
      // 2. 타이틀 캡션 변경
      if (showcaseCaption) {
        showcaseCaption.textContent = project.caption || project.title;
      }
  
      // 3. 리스트 활성화 상태 갱신
      projectItems.forEach((item, idx) => {
        if (idx === currentProjectIdx) {
          item.classList.add('active');
        } else {
          item.classList.remove('active');
        }
      });

      // 4. [핵심] 화살표 표시 상태 즉시 업데이트
      updateArrowVisibility();
    }
  
    // 왼쪽 화살표 (<) 클릭
    if (prevBtn) {
      prevBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        if (currentProjectIdx <= 0) return; // 0번이면 뒤로 가지 않음

        let prevIdx = currentProjectIdx - 1;
        while (prevIdx >= 0 && !portfolioData[prevIdx].video) prevIdx--;
        
        if (prevIdx >= 0) {
          playProject(prevIdx);
        }
      });
    }
  
    // 오른쪽 화살표 (>) 클릭
    if (nextBtn) {
      nextBtn.addEventListener('click', (e) => {
        e.stopPropagation();
        
        // 영상이 있는 다음 프로젝트 인덱스 찾기
        let nextIdx = currentProjectIdx + 1;
        while (nextIdx < portfolioData.length && !portfolioData[nextIdx].video) {
          nextIdx++;
        }

        // 유효한 다음 프로젝트가 있다면 재생
        if (nextIdx < portfolioData.length) {
          playProject(nextIdx);
        }
      });
    }
  
    // 왼쪽 목록 클릭 시
    projectItems.forEach((item, idx) => {
      item.addEventListener('click', (e) => {
        if (e.target.classList.contains('btn-storyboard')) return;
        playProject(idx);
      });
    });

    // 페이지 처음 열렸을 때 0번 상태 반영 (왼쪽 화살표 숨김)
    updateArrowVisibility();
  
    // 메인 비디오 클릭 시 전체화면 크게 보기
    if (mainViewTrigger && fullViewer && viewerBox) {
    mainViewTrigger.addEventListener('click', () => {
      const currentSrc = showcaseVideo.currentSrc || showcaseVideo.src || showcaseVideo.querySelector('source')?.src;
      if (!currentSrc) return;

      viewerBox.innerHTML = `
        <video id="popupVideoPlayer" src="${currentSrc}" controls autoplay playsinline style="max-width: 100%; max-height: 100%; border-radius: 12px;"></video>
      `;
      
      if (showcaseVideo) showcaseVideo.pause();
      pauseBGM(); // 전체화면 팝업이 뜨면 BGM 즉시 정지

      fullViewer.style.display = 'flex';
    });
  }

  // 2) 전체화면 뷰어 닫기 버튼 클릭 시
  if (closeViewerBtn) {
    closeViewerBtn.addEventListener('click', () => {
      fullViewer.style.display = 'none';
      viewerBox.innerHTML = '';
      
      // 2페이지면 쇼케이스 영상 재생(이때 BGM은 자동 정지 유지), 다른 페이지면 BGM 재생
      if (currentIndex === 1 && showcaseVideo) {
        showcaseVideo.play().catch(() => {});
      } else {
        playBGM();
      }
    });
  }
  
    // 스토리보드 버튼 클릭 (PDF 또는 이미지 팝업)
    document.querySelectorAll('.btn-storyboard').forEach((btn, idx) => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
  
        const file = portfolioData[idx]?.storyboard || btn.getAttribute('data-storyboard');
        if (!file) return alert('등록된 스토리보드 파일이 없습니다.');
  
        const safeUrl = encodeURI(file);
        const isImg = /\.(jpg|jpeg|png|webp|gif)$/i.test(file);
  
        if (isImg) {
          viewerBox.innerHTML = `
            <div style="width: 100%; height: 100%; display: flex; align-items: center; justify-content: center; overflow: auto;">
              <img src="${safeUrl}" alt="스토리보드" style="max-width: 100%; max-height: 100%; object-fit: contain; border-radius: 8px;" />
            </div>
          `;
        } else {
          viewerBox.innerHTML = `
            <object data="${safeUrl}" type="application/pdf" width="100%" height="100%">
              <embed src="${safeUrl}" type="application/pdf" width="100%" height="100%" />
              <div style="color: #fff; text-align: center; padding: 20px;">
                <p style="margin-bottom: 12px;">브라우저 미리보기가 지원되지 않습니다.</p>
                <a href="${safeUrl}" target="_blank" style="color: #4da3ff; text-decoration: underline;">새 창에서 열기</a>
              </div>
            </object>
          `;
        }
  
        if (showcaseVideo) showcaseVideo.pause();
        fullViewer.style.display = 'flex';
      });
    });
  
    /* ==================================================
       4. 하단 슬라이더 클릭 -> 카드 확장 모달 (FLIP 모션)
       ================================================== */
       document.querySelectorAll('.slider-item').forEach((item, idx) => {
      item.addEventListener('click', (e) => {
        e.stopPropagation();

        // 등록된 2개 프로젝트 인덱스(0, 1)로 자동 매핑
        const projectIdx = idx % 2; 

        // 1. 해당 프로젝트 영상 및 캡션/리스트 활성화
        playProject(projectIdx);

        // 2. 2페이지(Portfolio)로 부드럽게 스크롤 이동
        moveToSection(1);
      });
    });
  
    /* ==================================================
       5. 전체화면 뷰어 닫기 및 초기화
       ================================================== */
    if (closeViewerBtn) {
      closeViewerBtn.addEventListener('click', () => {
        fullViewer.style.display = 'none';
        viewerBox.innerHTML = '';
        // 전체화면 팝업을 닫았을 때 메인 쇼케이스 영상 다시 재생
        if (showcaseVideo) showcaseVideo.play().catch(() => {});
      });
    }
  
    window.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') {
        closeFlyCard();
        if (fullViewer) fullViewer.style.display = 'none';
        if (viewerBox) viewerBox.innerHTML = '';
        if (showcaseVideo) showcaseVideo.play().catch(() => {});
      }
    });
